# 注意只有chain id一致时Pymol才能计算RMSD!!!

PyMOL>alter ICL1 and chain R, chain='A'  Alter: modified 271 atoms.
PyMOL>alter ICL2 and chain R, chain='A'  Alter: modified 166 atoms.
PyMOL>alter ICL3 and chain R, chain='A'  Alter: modified 200 atoms.

PyMOL>alter ICL1_SM, chain='A'  Alter: modified 105 atoms.
PyMOL>alter ICL2_SM, chain='A'  Alter: modified 166 atoms.
PyMOL>alter ICL3_SM, chain='A'  Alter: modified 200 atoms.
